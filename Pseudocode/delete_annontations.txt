START

    IMPORT user_auth
    IMPORT config

    SET response to a dictionary with 'success' as False and 'message' as empty string

    IF 'annotationId' or 'uploadId' is not present in POST request
        SET response['message'] to "Missing required parameters"
        OUTPUT response as JSON
        EXIT
    END IF

    SET annotationId to POST['annotationId']
    SET uploadId to POST['uploadId']
    SET userId to SESSION['user_id']

    TRY
        PREPARE SQL statement to delete from annotation table where AnnotationID, UserID, and UploadID match
        BIND annotationId, userId, and uploadId as integers to the SQL statement

        IF SQL statement executes successfully
            SET response['success'] to True
            SET response['message'] to "Annotation deleted successfully"
        ELSE
            SET response['message'] to "Error deleting annotation"
        END IF
    CATCH Exception as e
        SET response['message'] to "Database error: " + exception message
    END TRY

    OUTPUT response as JSON

END
