//delete_annotatitons.php
START

    IMPORT user_auth
    IMPORT config

    SET response to a dictionary with 'success' as False and 'message' as empty string

    IF 'annotationId' or 'uploadId' is not present in POST request
        SET response['message'] to "Missing required parameters"
        OUTPUT response as JSON
        EXIT
    END IF

    SET annotationId to POST['annotationId']
    SET uploadId to POST['uploadId']
    SET userId to SESSION['user_id']

    TRY
        PREPARE SQL statement to delete from annotation table where AnnotationID, UserID, and UploadID match
        BIND annotationId, userId, and uploadId as integers to the SQL statement

        IF SQL statement executes successfully
            SET response['success'] to True
            SET response['message'] to "Annotation deleted successfully"
        ELSE
            SET response['message'] to "Error deleting annotation"
        END IF
    CATCH Exception as e
        SET response['message'] to "Database error: " + exception message
    END TRY

    OUTPUT response as JSON

END





//edit_annotation.php
START

    IMPORT user authentication module
    IMPORT database configuration

    INITIALIZE response as a dictionary with 'success' set to False and 'message' as an empty string

    IF any of the required POST parameters ('annotationId', 'uploadId', 'dataDate', 'note') are missing THEN
        SET response['message'] to "Missing required parameters"
        OUTPUT response as JSON
        EXIT
    END IF

    SET annotationId to integer value of POST['annotationId']
    SET uploadId to integer value of POST['uploadId']
    SET userId to the current session's user_id
    SET dataDate to POST['dataDate']
    SET note to POST['note']

    TRY
        PREPARE SQL statement to update annotation:
            "UPDATE annotation SET DataDate = ?, AnnotationText = ? WHERE AnnotationID = ? AND UserID = ? AND UploadID = ?"
        BIND parameters (dataDate, note, annotationId, userId, uploadId) to the SQL statement

        IF statement executes successfully THEN
            IF affected rows > 0 THEN
                SET response['success'] to True
                SET response['message'] to "Annotation updated successfully"
            ELSE
                SET response['message'] to "No changes made or annotation not found"
            END IF
        ELSE
            SET response['message'] to "Error updating annotation"
        END IF
    CATCH any Exception as e
        SET response['message'] to "Database error: " + e.message
    END TRY

    SET HTTP header to 'Content-Type: application/json'
    OUTPUT response as JSON

END





//get_annotations.php
START

    REQUIRE user authentication from '../auth/user_auth.php'
    REQUIRE database configuration from '../config.php'

    SET userId to the current user's ID from session
    SET uploadId to the value from GET parameter 'uploadId'

    PREPARE SQL statement:
        "SELECT AnnotationID, DataDate, AnnotationText FROM annotation WHERE UserID = ? AND UploadID = ?"
    BIND userId and uploadId as parameters to the SQL statement
    EXECUTE the SQL statement

    GET the result set from the executed statement

    INITIALIZE an empty list called annotations

    FOR each row in the result set
        CREATE an annotation dictionary with:
            "id" set to AnnotationID from the row
            "date" set to DataDate from the row
            "note" set to AnnotationText from the row
        APPEND the annotation dictionary to the annotations list
    END FOR

    OUTPUT the annotations list as a JSON response

END
