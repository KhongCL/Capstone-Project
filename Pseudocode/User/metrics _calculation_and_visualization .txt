//functions.php
START

    IMPORT necessary classes and libraries

    DEFINE FUNCTION handleCsvUpload WITH PARAMETERS conn, file
        LOG "HANDLE CSV UPLOAD START"
        IF session indicates sample data is being used
            CLEAR all sample data session variables
            CLEAR cached metrics and traffic sources from session
        END IF

        IF file upload has error
            RETURN error message with upload error
        END IF

        IF file is not a CSV
            RETURN error message about invalid file type
        END IF

        IF file size exceeds 5MB
            RETURN error message about file size
        END IF

        ENSURE config directory and csv_mappings.json exist
        ENSURE uploads directory exists

        GENERATE unique filename for upload
        MOVE uploaded file to uploads directory

        STORE file info in session

        TRY
            CREATE CsvProcessor instance
            EXTRACT metadata from CSV file
            PROCESS the CSV file for format and mapping

            IF processing is successful
                TRANSFORM data using mapping
                STORE metadata and file info in session
                SAVE transformed data to database

                IF save is successful
                    CLEAR upload and comparison session data
                    RETURN success message
                ELSE
                    CLEAR relevant session data on failure
                    DELETE uploaded file
                    RETURN error message from save
                END IF

            ELSE IF mapping is needed
                STORE mapping info in session
                CLEAR sample and comparison session data if not in comparison context
                IF AJAX request
                    RETURN needs_mapping with redirect info
                ELSE
                    REDIRECT to mapping page
                END IF

            ELSE
                DELETE uploaded file
                RETURN error message from processing
            END IF

        CATCH any exception
            DELETE uploaded file
            LOG error
            RETURN error message
        END TRY
    END FUNCTION

    DEFINE FUNCTION saveTransformedData WITH PARAMETERS conn, transformedData
        IF transformedData is empty
            IF validation errors exist in session
                RETURN error message with all validation errors
            ELSE
                CLEAR relevant session data
                RETURN error about no valid data
            END IF
        END IF

        TRY
            BEGIN database transaction
            GET user ID from session
            GET metadata from session
            INSERT new CSV_UPLOAD record with metadata

            FOR each row in transformedData
                GET or create source type ID
                FOR each metric in row
                    IF value is valid
                        INSERT data point into PROCESSED_DATA_POINT
                    END IF
                END FOR
            END FOR

            COMMIT transaction
            CLEAR cached metrics from session
            RETURN success message

        CATCH any exception
            ROLLBACK transaction
            RETURN error message
        END TRY
    END FUNCTION

    DEFINE FUNCTION getKeyMetrics WITH PARAMETERS conn, uploadId (optional)
        IF uploadId not provided
            GET user ID from session
            GET current upload ID using getCurrentUploadId
        END IF

        IF uploadId is not found
            RETURN default metrics
        END IF

        QUERY for total page views (Sessions, Page Views, visits)
        QUERY for unique visitors (Users, Unique visitors, Engaged sessions)
        QUERY for average session duration
        QUERY for bounce rate (calculated from engagement rate)

        RETURN metrics as dictionary
    END FUNCTION

    DEFINE FUNCTION getTrafficOverTime WITH PARAMETERS conn, interval, uploadId (optional)
        IF uploadId not provided
            GET user ID from session
            GET current upload ID using getCurrentUploadId
        END IF

        IF uploadId is not found
            RETURN empty data
        END IF

        QUERY for sessions data grouped by date
        RETURN list of traffic data by time period
    END FUNCTION

    DEFINE FUNCTION getTrafficSourcesDistribution WITH PARAMETERS conn, uploadId (optional)
        IF uploadId not provided
            GET user ID from session
            GET current upload ID using getCurrentUploadId
        END IF

        IF uploadId is not found
            RETURN empty data
        END IF

        QUERY for traffic sources and visit counts
        CALCULATE percentage for each source
        RETURN list of sources with counts and percentages
    END FUNCTION

    DEFINE FUNCTION getTopVisitedPages WITH PARAMETERS conn, limit
        GET user ID from session
        GET current upload ID using getCurrentUploadId

        IF uploadId is not found
            RETURN empty data
        END IF

        QUERY for top pages by sessions
        ESTIMATE unique visitors
        STORE data quality info in session
        RETURN list of top pages
    END FUNCTION

    DEFINE FUNCTION getCurrentUploadId WITH PARAMETERS conn, userId
        IF session indicates sample data is being used
            VERIFY sample upload exists in database
            IF valid
                RETURN sample upload ID
            END IF
        END IF

        QUERY for most recent user upload
        RETURN upload ID or null
    END FUNCTION

    DEFINE FUNCTION isUsingSampleData
        RETURN True if session indicates sample data, else False
    END FUNCTION

    DEFINE FUNCTION getSampleDataNotice
        IF session indicates sample data
            RETURN notice message and action link
        ELSE
            RETURN no notice
        END IF
    END FUNCTION

END

//traffic_sources.php
START

    REQUIRE authentication and configuration files
    INCLUDE functions file

    // Handle sample data session restoration
    IF 'sample_data' is set in GET parameters AND equals '1'
        IF session is not active
            START session
        END IF

        IF session variables for sample data are missing
            LOG warning about missing session variables
            QUERY database for latest sample data upload
            IF found
                SET session variables for sample data and upload ID
                LOG restoration of sample data session
            END IF
        END IF
    END IF

    // Determine user role and navigation
    IF session role is 'Admin'
        SET isAdmin to True
        SET backUrl to admin sample data page
    ELSE
        SET isAdmin to False
        SET backUrl to user index page
    END IF

    SET page title and active page variables

    // Get current upload ID (sample-aware)
    CALL getCurrentUploadId with database connection and user ID
    STORE result in uploadId

    // Get sample data notice
    CALL getSampleDataNotice
    STORE result in sampleNotice

    // Get traffic sources data for current upload
    CALL getTrafficSourcesDistribution with database connection and uploadId
    STORE result in sourcesData

    // Render HTML page
    OUTPUT HTML header and styles

    IF isAdmin
        OUTPUT admin header
    ELSE
        INCLUDE user header
    END IF

    OUTPUT main section with:
        - Page title
        - Filter panel with:
            - Top sources dropdown
            - Minimum percentage input
            - Quick filter buttons (major, moderate, minor, clear)
            - Source selection checkboxes (populated by JS)
            - Filter summary area
        - Sample data notice (if applicable)
        - Chart section for traffic source distribution
            - Chart type toggle (pie/bar)
            - Export to PDF button
        - Data table section for traffic sources breakdown
            - Table with source, visits, percentage columns
            - Select all/row selection checkboxes
            - Export to CSV button

    IF not isAdmin
        INCLUDE user footer
    END IF

    // JavaScript logic for metrics calculation and visualization

    PARSE sourcesData, uploadId, isSampleData from PHP to JS

    EXTRACT labels, visitCounts, percentages from sourcesData

    DEFINE background colors for charts

    INITIALIZE filteredData as a copy of sourcesData
    INITIALIZE selectedSources as empty set
    INITIALIZE currentFilters with default values

    DEFINE function createChart(type)
        IF chart exists, destroy it
        GET filtered chart data
        IF no data
            HIDE chart, SHOW "No Data" message
            RETURN
        SHOW chart, HIDE "No Data" message
        CONFIGURE chart (pie/bar) with filtered data and options
        RENDER chart
    END FUNCTION

    DEFINE function getFilteredChartData()
        RETURN labels, visitCounts, percentages from filteredData
    END FUNCTION

    DEFINE function getFilteredData()
        COPY sourcesData to data
        APPLY top sources filter if set
        APPLY minimum percentage filter if set
        APPLY selected sources filter if set
        RECALCULATE percentages for filtered data
        RETURN data
    END FUNCTION

    // Chart type toggle event
    ON chart type button click
        CALL createChart with selected type
        UPDATE active button state
    END EVENT

    CALL createChart('pie') to initialize

    // Filter panel logic
    CALL initializeFilters()
        POPULATE source selection checkboxes
        ADD event listeners for filters
        SELECT all sources by default
    END FUNCTION

    DEFINE function populateSourceSelection()
        FOR each source in sourcesData
            CREATE checkbox for source
            ADD to source selection container
        END FOR
    END FUNCTION

    DEFINE function handleTopSourcesChange()
        UPDATE currentFilters.topSources
        CALL applyFilters()
    END FUNCTION

    DEFINE function handleMinPercentageChange()
        UPDATE currentFilters.minPercentage
        CALL applyFilters()
    END FUNCTION

    DEFINE function handleSourceSelectionChange()
        UPDATE currentFilters.selectedSources from checked boxes
        UPDATE table row selection
        CALL applyFilters()
    END FUNCTION

    DEFINE function applyQuickFilter(type)
        RESET all filters
        SET currentFilters.selectedSources based on type (major, moderate, minor)
        UPDATE source checkboxes
        CALL applyFilters()
        IF no sources selected
            ALERT user no data found for filter
        END IF
    END FUNCTION

    DEFINE function clearAllFilters()
        RESET all filters to default
        SELECT all sources
        CALL applyFilters()
    END FUNCTION

    DEFINE function selectAllSources()
        CHECK all source checkboxes
        UPDATE currentFilters.selectedSources to all sources
        UPDATE table row selection
        CALL applyFilters()
    END FUNCTION

    DEFINE function deselectAllSources()
        UNCHECK all source checkboxes
        CLEAR currentFilters.selectedSources
        UPDATE table row selection
        CALL applyFilters()
    END FUNCTION

    DEFINE function updateSourceCheckboxes()
        SET checked state of source checkboxes based on currentFilters.selectedSources
        UPDATE table row selection
    END FUNCTION

    DEFINE function applyFilters()
        UPDATE filteredData using getFilteredData()
        UPDATE table, chart, and filter summary
    END FUNCTION

    DEFINE function updateTable()
        FOR each table row
            DETERMINE if row should be shown based on filters
            SHOW or HIDE row accordingly
        END FOR
        UPDATE no data message for table
        UPDATE table percentages for visible rows
    END FUNCTION

    DEFINE function updateTableNoDataMessage(show)
        SHOW or HIDE "No Data" message in table section
    END FUNCTION

    DEFINE function updateTablePercentages()
        FOR each visible row
            UPDATE percentage cell based on filteredData
        END FOR
    END FUNCTION

    DEFINE function updateChart()
        GET current chart type
        CALL createChart with current type
    END FUNCTION

    DEFINE function updateFilterSummary()
        IF no data or all data shown
            HIDE summary
        ELSE
            SHOW summary with count and percentage of visits
        END IF
    END FUNCTION

    // Table row selection logic
    DEFINE function toggleRowSelection(row)
        TOGGLE checkbox and row selected class
        UPDATE source checkboxes from table
    END FUNCTION

    DEFINE function toggleAllRows(masterCheckbox)
        SET all row checkboxes to match master
        UPDATE row selected classes
        UPDATE source checkboxes from table
    END FUNCTION

    DEFINE function updateTableRowSelection()
        FOR each row
            SET checkbox and row class based on currentFilters.selectedSources
        END FOR
        UPDATE master checkbox state
    END FUNCTION

    DEFINE function updateSourceCheckboxFromTable()
        UPDATE currentFilters.selectedSources from checked table rows
        UPDATE source checkboxes
        UPDATE master checkbox
        CALL applyFilters()
    END FUNCTION

    DEFINE function updateMasterCheckbox()
        SET master checkbox checked/indeterminate based on row selection
    END FUNCTION

    // Export logic
    DEFINE function exportTableToCSV()
        IF no filteredData
            ALERT user
            RETURN
        CREATE CSV with headers and filteredData
        TRIGGER download with descriptive filename
        LOG export in database via AJAX
    END FUNCTION

    DEFINE async function exportChartToPDF()
        IF no filteredData
            ALERT user
            RETURN
        CREATE PDF with:
            - Title, generated info, sample data notice
            - Filter info (if filtered)
            - Traffic sources summary (totals, top source)
            - Table of filtered sources
            - Pie and bar charts (captured as images)
            - Key insights (top 3 sources, diversity)
            - Report info (upload ID, type, etc.)
        SAVE PDF with descriptive filename
        LOG export in database via AJAX
    END FUNCTION

END





//get_traffic_data.php
START
    IMPORT user authentication module from '../auth/user_auth.php'
    IMPORT database configuration from '../config.php'
    IMPORT functions from '../functions.php'

    SET the HTTP response header to 'Content-Type: application/json'

    IF 'interval' parameter exists in the GET request
        SET interval to the value of 'interval' from GET request
    ELSE
        SET interval to 'day'
    END IF

    IF interval is not one of ['hour', 'day', 'month', 'year']
        SET interval to 'day'
    END IF

    CALL getTrafficOverTime WITH database connection and interval
        STORE the result in trafficData

    CONVERT trafficData to JSON format
    OUTPUT the JSON data as the HTTP response
END
