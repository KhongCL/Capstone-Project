
//login.php
START
    IMPORT session
    IMPORT config.php

    INITIALIZE errors as empty dictionary

    IF session contains 'user_id' THEN
        IF session 'role' is 'Admin' THEN
            REDIRECT to "admin/index.php"
        ELSE
            REDIRECT to "user/index.php"
        END IF
        EXIT
    END IF

    IF request method is POST THEN
        SET username to trimmed POST 'username' or empty string
        SET password to POST 'password' or empty string

        IF username is empty THEN
            SET errors['username'] to "Username is required"
        END IF

        IF password is empty THEN
            SET errors['password'] to "Password is required"
        END IF

        IF errors is empty THEN
            PREPARE SQL statement to SELECT UserID, Username, PasswordHash, Role, AccountStatus FROM user WHERE Username = ?
            BIND username to SQL statement
            EXECUTE SQL statement
            GET result

            IF user is found in result THEN
                IF user's AccountStatus is "Suspended" THEN
                    SET errors['general'] to "Your account has been suspended. Please contact support."
                ELSE IF password matches user's PasswordHash THEN
                    SET session 'user_id' to user's UserID
                    SET session 'username' to user's Username
                    SET session 'role' to user's Role

                    IF user's Role is "Admin" THEN
                        REDIRECT to "admin/index.php"
                    ELSE
                        REDIRECT to "user/index.php"
                    END IF
                    EXIT
                ELSE
                    SET errors['general'] to "Invalid username or password"
                END IF
            ELSE
                SET errors['general'] to "Invalid username or password"
            END IF
        END IF
    END IF

    DISPLAY HTML login form
        IF errors['general'] exists THEN
            DISPLAY error message
        END IF
        DISPLAY username and password input fields
        DISPLAY "Remember me" checkbox
        DISPLAY submit button
        DISPLAY link to registration page

    INCLUDE JavaScript for:
        - Removing error bubbles on input
        - Showing error bubbles for server-side errors
        - Toggling password visibility

    DISPLAY overlay and popup for login success (if needed)
END





//register.php
START

    START SESSION
    IMPORT config.php

    IF user is already logged in THEN
        IF user role is 'Admin' THEN
            REDIRECT to admin dashboard
        ELSE
            REDIRECT to user dashboard
        END IF
        EXIT
    END IF

    INITIALIZE errors as empty list
    INITIALIZE success as False

    IF request method is POST THEN
        GET and trim firstname from POST or set to empty string
        GET and trim lastname from POST or set to empty string
        GET and trim username from POST or set to empty string
        GET and trim email from POST or set to empty string
        GET password from POST or set to empty string
        GET confirmPassword from POST or set to empty string
        SET terms to True if terms is set in POST, else False

        IF firstname is empty THEN
            ADD "First name is required" to errors
        END IF

        IF lastname is empty THEN
            ADD "Last name is required" to errors
        END IF

        IF username is empty THEN
            ADD "Username is required" to errors
        ELSE IF username length < 3 THEN
            ADD "Username must be at least 3 characters" to errors
        END IF

        IF email is empty THEN
            ADD "Email is required" to errors
        ELSE IF email is not valid THEN
            ADD "Please enter a valid email address" to errors
        END IF

        IF password is empty THEN
            ADD "Password is required" to errors
        ELSE IF password length < 8 THEN
            ADD "Password must be at least 8 characters" to errors
        END IF

        IF password does not match confirmPassword THEN
            ADD "Passwords do not match" to errors
        END IF

        IF terms is not checked THEN
            ADD "You must agree to the terms and conditions" to errors
        END IF

        IF errors is empty THEN
            PREPARE SQL to SELECT user by username or email
            EXECUTE query
            IF user exists THEN
                PREPARE SQL to SELECT user by username
                EXECUTE query
                IF username exists THEN
                    ADD "Username already exists" to errors
                END IF

                PREPARE SQL to SELECT user by email
                EXECUTE query
                IF email exists THEN
                    ADD "Email address already exists" to errors
                END IF
            ELSE
                HASH password
                PREPARE SQL to INSERT new user with username, email, hashed password, role 'End-User', status 'Active'
                EXECUTE query
                IF insert successful THEN
                    SET success to True
                    SET session register_success to True
                ELSE
                    ADD "Registration failed. Please try again." to errors
                END IF
            END IF
        END IF
    END IF

    OUTPUT HTML registration form with:
        - First name, last name, username, email, password, confirm password, terms checkbox
        - Client-side JS for password toggle and validation
        - Show server-side errors as bubbles on relevant fields
        - On successful registration, show popup and redirect to login page after 3 seconds

END
