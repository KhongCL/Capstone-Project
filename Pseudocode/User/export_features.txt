//Overview.php
START

    IMPORT datetime

    DEFINE FUNCTION export_data_table_to_csv WITH PARAMETERS data_table, filename
        OPEN the file named filename in write mode
            WRITE the column headers of data_table to the file, separated by commas
            FOR each row in data_table
                WRITE the row values to the file, separated by commas
            END FOR
        CLOSE the file
        OUTPUT "Data table exported to CSV successfully."
    END FUNCTION

    DEFINE FUNCTION export_dashboard_view_to_csv WITH PARAMETERS dashboard_data, filename
        OPEN the file named filename in write mode
            WRITE "Dashboard Export" and the current date/time to the file
            FOR each section in dashboard_data
                WRITE the section title to the file
                IF the section contains a table
                    WRITE the table headers to the file
                    FOR each row in the table
                        WRITE the row values to the file, separated by commas
                    END FOR
                ELSE IF the section contains a metric or value
                    WRITE the metric name and value to the file
                END IF
                WRITE a blank line to separate sections
            END FOR
        CLOSE the file
        OUTPUT "Dashboard view exported to CSV successfully."
    END FUNCTION

    DEFINE FUNCTION export_dashboard_view_to_pdf WITH PARAMETERS dashboard_data, filename
        INITIALIZE a new PDF document
        ADD a title "Dashboard Export" and the current date/time to the PDF
        FOR each section in dashboard_data
            ADD the section title to the PDF
            IF the section contains a chart
                CAPTURE the chart as an image
                ADD the chart image to the PDF
            END IF
            IF the section contains a table
                ADD the table headers to the PDF
                FOR each row in the table
                    ADD the row values to the PDF
                END FOR
            ELSE IF the section contains a metric or value
                ADD the metric name and value to the PDF
            END IF
            ADD a blank line or spacing to separate sections
        END FOR
        SAVE the PDF document with the given filename
        OUTPUT "Dashboard view exported to PDF successfully."
    END FUNCTION

    DEFINE FUNCTION export_dashboard
        OUTPUT "Export Dashboard"
        OUTPUT "1. Export Data Table to CSV"
        OUTPUT "2. Export Dashboard View to CSV"
        OUTPUT "3. Export Dashboard View to PDF"
        OUTPUT "4. Exit"

        WHILE True
            ASK the user to enter their choice as a number
            IF the input is a valid number THEN
                IF the choice is 1
                    CALL export_data_table_to_csv WITH the current data table and "data_table_export.csv"
                ELSE IF the choice is 2
                    CALL export_dashboard_view_to_csv WITH the current dashboard data and "dashboard_export.csv"
                ELSE IF the choice is 3
                    CALL export_dashboard_view_to_pdf WITH the current dashboard data and "dashboard_export.pdf"
                ELSE IF the choice is 4
                    OUTPUT "Exiting export menu."
                    BREAK
                ELSE
                    OUTPUT "Invalid choice. Please enter 1, 2, 3, or 4."
                END IF
            ELSE
                OUTPUT "Invalid input. Please enter a number."
            END IF
        END WHILE
    END FUNCTION

END




//traffic_sources.php
START

    IMPORT datetime

    DEFINE FUNCTION export_table_to_csv WITH PARAMETERS table_data, filename
        OPEN the file named filename in write mode
            FOR each row in table_data
                JOIN the row values with commas
                WRITE the joined row followed by a newline to the file
            END FOR
        CLOSE the file
        OUTPUT "Table exported to CSV successfully as " + filename
    END FUNCTION

    DEFINE FUNCTION export_dashboard_to_pdf WITH PARAMETERS dashboard_data, filename
        // dashboard_data may include charts, tables, and summary info
        INITIALIZE a new PDF document
        SET page_number to 1

        // Add dashboard title and date
        ADD "Dashboard Report" as title to the PDF
        ADD "Generated on: " + current date and time to the PDF

        // Add summary section if available
        IF dashboard_data contains summary
            ADD summary section to the PDF
        END IF

        // Add charts if available
        IF dashboard_data contains charts
            FOR each chart in dashboard_data.charts
                ADD chart image or representation to the PDF
            END FOR
        END IF

        // Add tables if available
        IF dashboard_data contains tables
            FOR each table in dashboard_data.tables
                ADD table title to the PDF
                FOR each row in table
                    ADD row data to the PDF as a table row
                END FOR
            END FOR
        END IF

        SAVE the PDF document as filename
        OUTPUT "Dashboard exported to PDF successfully as " + filename
    END FUNCTION

    DEFINE FUNCTION get_dashboard_table_data
        // Retrieve or generate the data table to be exported
        RETURN table_data
    END FUNCTION

    DEFINE FUNCTION get_dashboard_view_data
        // Retrieve or generate all dashboard data (charts, tables, summary)
        RETURN dashboard_data
    END FUNCTION

    DEFINE FUNCTION dashboard_export_menu
        OUTPUT "Dashboard Export Menu"
        OUTPUT "1. Export Table to CSV"
        OUTPUT "2. Export Dashboard to PDF"
        OUTPUT "3. Exit"

        WHILE True
            ASK the user to enter their choice as a number
            IF the input is a valid number THEN
                IF the choice is 1
                    CALL get_dashboard_table_data AND STORE in table_data
                    ASK the user to enter a filename for the CSV export
                    CALL export_table_to_csv WITH table_data, filename
                ELSE IF the choice is 2
                    CALL get_dashboard_view_data AND STORE in dashboard_data
                    ASK the user to enter a filename for the PDF export
                    CALL export_dashboard_to_pdf WITH dashboard_data, filename
                ELSE IF the choice is 3
                    OUTPUT "Exiting Dashboard Export Menu."
                    RETURN
                ELSE
                    OUTPUT "Invalid choice. Please enter 1, 2, or 3."
                END IF
            ELSE
                OUTPUT "Invalid input. Please enter a number."
            END IF
        END WHILE
    END FUNCTION

    // Entry point for dashboard export features
    CALL dashboard_export_menu

END




//pages.php
Start

DEFINE FUNCTION export_table_to_csv
    OUTPUT "Exporting table to CSV..."
    GET the table element from the page
    INITIALIZE an empty list called csv_rows

    FOR each row in the table
        INITIALIZE an empty list called cols
        FOR each cell in the row
            ADD the cell's text (trimmed and quoted) to cols
        END FOR
        JOIN cols with commas and ADD to csv_rows
    END FOR

    JOIN csv_rows with newline characters to form csv_content
    CREATE a CSV file blob from csv_content
    CREATE a download link for the CSV file
    TRIGGER the download for the user

    CALL log_export WITH parameters: export_type="CSV", description="Exported top pages table data"
END FUNCTION

DEFINE FUNCTION export_chart_to_pdf
    OUTPUT "Exporting chart and table to PDF..."
    INITIALIZE a new PDF document

    GET the current date and time as generated_date
    GET the username from the session

    ADD a header to the PDF with the report title, generated_date, and username

    IF sample data is used
        ADD a sample data warning to the PDF
    END IF

    ADD a summary section to the PDF with:
        - Total pages analyzed
        - Total page views
        - Total unique visitors
        - Average views per visitor

    ADD a table to the PDF with columns:
        - Page URL
        - Page Views
        - Unique Visitors
        - Views/Visitor

    FOR each page in the data
        ADD a row to the PDF table with the page's data
    END FOR

    ADD a chart image to the PDF by capturing the chart element as an image

    ADD a "Key Insights" section with the top 3 most visited pages

    IF data quality is estimated
        ADD a data quality notice to the PDF
    END IF

    ADD a "Report Information" section with:
        - Upload ID
        - Report type
        - Data source
        - Totals

    IF sample data is used
        ADD a note about sample data to the PDF
    END IF

    SAVE the PDF file with a descriptive filename
    TRIGGER the download for the user

    CALL log_export WITH parameters: export_type="PDF", description="Exported top pages comprehensive report as PDF"
END FUNCTION

DEFINE FUNCTION log_export WITH PARAMETERS export_type, description
    SEND a POST request to the server endpoint "log_export.php" with export_type and description
    IF the server responds with success
        OUTPUT "Export logged successfully."
    ELSE
        OUTPUT "Failed to log export."
    END IF

END 




//log_export.php
START
    IMPORT config
    IMPORT user_auth

    IF the request method is POST THEN
        START session
        SET userId to the value of 'user_id' from session, or None if not set

        IF userId is not set THEN
            SET HTTP response code to 403
            OUTPUT JSON: success = False, message = "User not authenticated"
            EXIT
        END IF

        SET exportType to the value of 'exportType' from POST data, or "CSV" if not set
        SET description to the value of 'description' from POST data, or empty string if not set

        PREPARE SQL statement:
            "INSERT INTO export_history (UserID, ExportType, ExportedDataDescription) VALUES (?, ?, ?)"
        BIND userId, exportType, description to the SQL statement
        EXECUTE the SQL statement and store the result in success

        OUTPUT JSON: success = success
    ELSE
        SET HTTP response code to 405
        OUTPUT JSON: success = False, message = "Invalid request method"
    END IF
END





//compare.php
START
    DEFINE FUNCTION compareCSVFiles WITH PARAMETERS file1_path, file2_path
        CALL parseCSV WITH file1_path AND STORE RESULT IN data1
        CALL parseCSV WITH file2_path AND STORE RESULT IN data2

        IF data1 IS EMPTY OR data2 IS EMPTY
            THROW Exception "One or both CSV files are empty or invalid."
        END IF

        SET headers1 TO KEYS OF data1[0]
        SET headers2 TO KEYS OF data2[0]

        SET analytics_metrics TO [
            'traffic_source', 'sessions', 'engaged_sessions', 'engagement_rate',
            'average_engagement_time_per_session', 'events_per_session', 'event_count',
            'key_events', 'session_key_event_rate', 'total_revenue'
        ]

        INITIALIZE comparison DICTIONARY WITH:
            basic_metrics: row/column counts and differences
            headers: file1_headers, file2_headers, common_headers, unique_to_file1, unique_to_file2
            analytics_metrics: EMPTY DICTIONARY
            summary_comparison: EMPTY DICTIONARY
            data_sample: first 5 rows from each file

        SET common_headers TO MERGE OF common_headers, headers1, headers2
        INITIALIZE semantic_common AS EMPTY LIST

        FOR EACH metric IN analytics_metrics
            SET found_column TO CALL findMetricColumn WITH common_headers, metric

            IF found_column EXISTS
                SET col1 TO CALL findMetricColumn WITH headers1, metric
                SET col2 TO CALL findMetricColumn WITH headers2, metric

                IF col1 AND col2 EXIST
                    APPEND metric TO semantic_common
                    SET values1 TO COLUMN col1 FROM data1
                    SET values2 TO COLUMN col2 FROM data2

                    IF col1 OR col2 INDICATES CONVERSION (e.g., "_CONVERT_FROM_")
                        HANDLE CONVERSION LOGIC (e.g., engagement_rate <-> bounce_rate)
                    END IF

                    CLEAN AND CONVERT values1 AND values2 TO NUMERIC ARRAYS

                    IF numeric1 AND numeric2 ARE NOT EMPTY
                        SET stats1 TO CALL calculateStats WITH numeric1
                        SET stats2 TO CALL calculateStats WITH numeric2

                        CALCULATE percent_change BETWEEN stats1 AND stats2

                        ADD TO comparison.analytics_metrics[metric]:
                            column_name, file1_column, file2_column, file1_stats, file2_stats,
                            comparison: total_diff, avg_diff, percent_change, improvement
                    END IF
                END IF
            END IF
        END FOR

        SET comparison.summary_comparison TO CALL calculateSummaryComparison WITH data1, data2, comparison.analytics_metrics
        SET comparison.headers.semantic_common TO semantic_common
        SET comparison.headers.semantic_common_count TO LENGTH OF semantic_common

        RETURN comparison
    END FUNCTION

    // Export Features

    DEFINE FUNCTION exportTableToCSV WITH PARAMETERS table_data, filename
        OPEN filename IN WRITE MODE
        FOR EACH row IN table_data
            WRITE row AS CSV TO FILE
        END FOR
        CLOSE FILE
    END FUNCTION

    DEFINE FUNCTION exportDashboardToCSV WITH PARAMETERS dashboard_data, filename
        EXTRACT TABLES AND CHART DATA FROM dashboard_data
        CALL exportTableToCSV FOR EACH TABLE OR CHART DATA
    END FUNCTION

    DEFINE FUNCTION exportDashboardToPDF WITH PARAMETERS dashboard_view, filename
        INITIALIZE PDF DOCUMENT
        ADD CHARTS AND TABLES FROM dashboard_view TO PDF
        SAVE PDF AS filename
    END FUNCTION
END




//CsvProcessor.php
START

    IMPORT csv
    IMPORT pdf_library

    DEFINE FUNCTION export_table_to_csv WITH PARAMETERS table_data, filename
        OPEN filename in write mode
        FOR each row in table_data
            WRITE row as CSV to the file
        END FOR
        CLOSE the file
        OUTPUT "Table exported to CSV successfully."
    END FUNCTION

    DEFINE FUNCTION export_dashboard_to_csv WITH PARAMETERS dashboard_data, filename
        OPEN filename in write mode
        FOR each widget in dashboard_data
            IF widget is a table
                FOR each row in widget
                    WRITE row as CSV to the file
                END FOR
            ELSE IF widget is a chart
                WRITE chart summary or data points as CSV to the file
            END IF
        END FOR
        CLOSE the file
        OUTPUT "Dashboard view exported to CSV successfully."
    END FUNCTION

    DEFINE FUNCTION export_dashboard_to_pdf WITH PARAMETERS dashboard_data, filename
        INITIALIZE a new PDF document
        FOR each widget in dashboard_data
            IF widget is a table
                ADD table to PDF document
            ELSE IF widget is a chart
                ADD chart image or summary to PDF document
            END IF
        END FOR
        SAVE the PDF document as filename
        OUTPUT "Dashboard view exported to PDF successfully."
    END FUNCTION

    DEFINE FUNCTION export_menu
        OUTPUT "Export Menu"
        OUTPUT "1. Export Table to CSV"
        OUTPUT "2. Export Dashboard to CSV"
        OUTPUT "3. Export Dashboard to PDF"
        OUTPUT "4. Return to Main Menu"

        WHILE True
            ASK user for choice
            IF choice is 1
                CALL export_table_to_csv WITH selected table and filename
            ELSE IF choice is 2
                CALL export_dashboard_to_csv WITH dashboard data and filename
            ELSE IF choice is 3
                CALL export_dashboard_to_pdf WITH dashboard data and filename
            ELSE IF choice is 4
                RETURN
            ELSE
                OUTPUT "Invalid choice. Please try again."
            END IF
        END WHILE
    END FUNCTION

END
