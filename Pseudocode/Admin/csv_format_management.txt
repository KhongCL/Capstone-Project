//admin_mappings.php
START

    IMPORT necessary PHP files for authentication, configuration, and CSV processing

    SET mappingsFile to the path of the CSV mappings JSON file
    INITIALIZE message and error as empty strings

    IF the HTTP request is POST AND the form submission is for updating mappings THEN
        INITIALIZE mappings as an empty dictionary

        FOR each format submitted in the POST data
            GET and sanitize the format name
            IF format name is empty THEN
                CONTINUE to next format
            END IF

            GENERATE a format key by lowercasing, removing special characters, and replacing spaces with underscores
            IF format key is empty THEN
                CONTINUE to next format
            END IF

            INITIALIZE the format's mapping structure with detection columns, column mappings, and data types

            IF columns are provided for this format THEN
                FOR each column mapping
                    IF source column is not empty THEN
                        HANDLE custom or predefined target field
                        IF custom target field is chosen THEN
                            VALIDATE the custom field name
                            IF invalid THEN
                                SET error message and CONTINUE
                            END IF
                        END IF

                        HANDLE custom or predefined data type
                        IF custom data type is chosen THEN
                            VALIDATE the custom data type
                            IF invalid THEN
                                SET error message and CONTINUE
                            END IF
                        END IF

                        IF target field is not empty THEN
                            SAVE the mapping and data type for this source column
                        END IF
                    END IF
                END FOR
            END IF
        END FOR

        IF no errors THEN
            SAVE the mappings to the JSON file
            IF successful THEN
                SET success message
                TRY
                    UPDATE the database with the new mappings
                    UPDATE the metric types in the database
                CATCH any exception
                    SET error message
                END TRY
            ELSE
                SET error message for file save failure
            END IF
        END IF
    END IF

    DEFINE function to validate data type names (letters, numbers, underscores, length)
    DEFINE function to validate system field names (lowercase, numbers, underscores, length, no leading/trailing underscore)

    IF the mappings file does not exist THEN
        CREATE default mappings for GA4 traffic acquisition
        SAVE default mappings to the file
    END IF

    LOAD mappings from the JSON file

    DEFINE function to update database mappings:
        DELETE all existing mappings and formats from the database
        FOR each format in mappings
            INSERT format into csv_format table
            FOR each column mapping
                INSERT mapping into column_mapping table
            END FOR
        END FOR

    DEFINE function to update metric types:
        COLLECT all unique system fields from mappings
        FOR each system field
            IF not already in metric_type table THEN
                INSERT with description (predefined or custom)
            END IF
        END FOR

    DEFINE function to clean up orphaned metric types (not used in any mapping)

    // Prepare data for the HTML form:
    COLLECT all system fields (predefined, used in mappings, from database)
    COLLECT all data types (predefined and custom from mappings)

    RENDER HTML page:
        DISPLAY admin header and navigation
        DISPLAY instructions for managing CSV mappings

        IF there is a success message THEN
            DISPLAY success message
        END IF

        IF there is an error message THEN
            DISPLAY error message
        END IF

        DISPLAY form for managing CSV formats:
            FOR each format in mappings
                DISPLAY format name and detection columns
                DISPLAY table of column mappings:
                    FOR each mapping
                        DISPLAY source column, target field (dropdown), data type (dropdown), and remove button
                    END FOR
                DISPLAY buttons to add/remove columns and formats
            END FOR

            DISPLAY buttons to add new format, save all mappings, and upload sample data

        INCLUDE JavaScript for:
            - Dynamic form handling (add/remove formats/columns)
            - Custom field and data type validation
            - Real-time validation and duplicate checks
            - Collapsible format sections

        INCLUDE admin footer

END






//upload_sample_data.php
START

    REQUIRE admin authentication
    REQUIRE configuration and CSV processor classes

    SET isPostRedirect to False
    IF HTTP_REFERER is set THEN
        IF HTTP_REFERER contains 'upload_sample.php' THEN
            SET isPostRedirect to True
        END IF
    END IF

    // Get list of supported report types for dropdown
    INITIALIZE reportTypes as empty list
    EXECUTE SQL: SELECT DISTINCT ReportType FROM csv_upload ORDER BY ReportType
    FOR each row in result
        ADD row['ReportType'] to reportTypes
    END FOR

    // Check for existing sample data
    SET hasExistingSampleData to False
    SET existingSampleInfo to None
    SET sampleDataPreview to None
    SET csvHeaders to empty list

    PREPARE SQL: SELECT UploadID, FileName, AccountName, PropertyName, DataDateStart, DataDateEnd FROM csv_upload WHERE IsSampleData = 1 ORDER BY UploadDate DESC LIMIT 1
    EXECUTE statement
    IF a row is returned THEN
        SET hasExistingSampleData to True
        SET existingSampleInfo to row

        SET fileName to row['FileName']
        SET filePath to uploads directory + fileName

        IF file exists at filePath THEN
            OPEN file in read mode
            WHILE reading each line
                IF line is not empty AND does not start with '#' THEN
                    SET csvHeaders to CSV-parsed values of the line
                    BREAK
                END IF
            END WHILE

            INITIALIZE sampleDataPreview as empty list
            SET rowCount to 0
            WHILE reading each line AND rowCount < 10
                IF line is not empty THEN
                    PARSE line as CSV and ADD to sampleDataPreview
                    INCREMENT rowCount
                END IF
            END WHILE
            CLOSE file
        END IF
    END IF

    // Render HTML page
    DISPLAY HTML header and styles

    INCLUDE admin header

    DISPLAY "Upload Sample CSV Data" section

    IF session contains 'sample_upload_message' AND isPostRedirect is True THEN
        DISPLAY message (success or error)
        UNSET 'sample_upload_message' from session
    END IF

    IF hasExistingSampleData AND existingSampleInfo is not None THEN
        DISPLAY existing sample data info (file, account, property, date range)
        IF sampleDataPreview is not empty THEN
            DISPLAY preview section with:
                - CSV headers as table headers
                - Up to 10 rows of sample data
                - Actions: Download, View Dashboard, View Traffic Sources, View Top Pages
        END IF
    END IF

    DISPLAY form to upload new sample file:
        - File input (CSV only, max 5MB)
        - Report type dropdown (existing types + "Define New")
        - New report type field (shown if "Define New" selected)
        - Submit button

    DISPLAY section to clear all sample data with confirmation

    INCLUDE admin footer

    // JavaScript logic for enhanced UX
    ON document ready:
        - Show/hide new report type field based on dropdown
        - On form submit:
            - Validate file and fields
            - Show confirmation dialog if replacing data or clearing errors
            - Submit via AJAX, show progress, handle response (success, error, validation errors)
        - On "Clear Sample Data" button click:
            - Show confirmation, send AJAX request, handle response

        - Toggle sample data preview table on header click

        - Show browser confirmation if navigating away with error messages present

        - Utility functions for showing/hiding messages, validation errors, and progress

END





//upload_sample.php
START

    IMPORT required authentication, configuration, CSV processor, and utility functions

    IF user is not logged in as admin
        REDIRECT to login page
        EXIT
    END IF

    LOG "=== SAMPLE UPLOAD START ==="
    LOG POST and FILES data

    INITIALIZE response as a dictionary with 'success' = False and empty 'message'

    DETERMINE if the request is AJAX

    IF request method is POST
        LOG "Processing POST request for sample upload"

        // File validation
        IF no file uploaded OR file upload error
            SET response message to upload error
            LOG error
        ELSE IF file size > 5MB
            SET response message to "File size exceeds the 5MB limit."
            LOG error
        ELSE IF file extension is not 'csv'
            SET response message to "Only CSV files are allowed."
            LOG error
        ELSE
            // Get report type
            SET reportType from POST data
            IF reportType is 'new' AND newReportType is provided
                SET reportType to newReportType
            END IF
            LOG reportType

            IF reportType is empty
                SET response message to "Please select or provide a report type."
                LOG error
            ELSE
                TRY
                    CREATE uploads directory if it does not exist
                    GENERATE unique file name and path
                    LOG file path

                    IF move uploaded file to uploads directory is successful
                        LOG success

                        CREATE CsvProcessor instance
                        EXTRACT metadata from CSV file
                        LOG metadata

                        IF file is empty
                            SET response message to "The uploaded CSV file is empty."
                            LOG error
                            DELETE file
                        ELSE
                            LOG file size

                            PROCESS the CSV file using CsvProcessor
                            LOG process result

                            IF process result is 'success' OR 'needs_mapping'
                                LOG "File processed successfully, transforming data..."

                                IF result is 'success'
                                    TRANSFORM data using provided mapping and format
                                ELSE
                                    BUILD mapping from suggestions with confidence > 50
                                    LOG mapping
                                    TRANSFORM data using mapping
                                END IF

                                LOG count of transformed data

                                IF session is not started
                                    START session
                                END IF

                                IF validation errors exist in session AND no data was transformed
                                    FORMAT validation errors for response
                                    SET response message to validation error summary
                                    CLEAR validation errors from session
                                    DELETE file
                                    RETURN response (AJAX or redirect)
                                    EXIT
                                END IF

                                IF no transformed data
                                    SET response message to "No valid data rows found..."
                                    LOG error
                                    DELETE file
                                ELSE
                                    LOG sample transformed row
                                    LOG "Attempting to save sample data..."
                                    CALL saveSampleData with connection, data, file info, reportType, metadata

                                    IF save successful
                                        SET response success to True
                                        SET response message to "Sample data uploaded and processed successfully."
                                        LOG success
                                        // Do not delete file (keep for preview)
                                    ELSE
                                        SET response message to "Failed to save sample data to database."
                                        LOG error
                                        DELETE file
                                    END IF
                                END IF
                            ELSE
                                SET response message to "Failed to process CSV file: ..."
                                LOG error
                                DELETE file
                            END IF
                        END IF
                    ELSE
                        SET response message to "Failed to save uploaded file."
                        LOG error
                    END IF
                CATCH Exception as e
                    SET response message to "Error processing file: " + exception message
                    LOG exception and stack trace

                    IF session is not started
                        START session
                    END IF

                    IF validation errors exist in session
                        FORMAT validation errors for response
                        SET response message to validation error summary
                        CLEAR validation errors from session
                    ELSE IF exception message indicates validation or parsing error
                        PARSE and format validation errors for response
                    END IF

                    DELETE file if it exists
                END TRY
            END IF
        END IF
    END IF

    LOG final response
    LOG "=== SAMPLE UPLOAD END ==="

    IF AJAX request
        RETURN response as JSON
    ELSE
        STORE response in session
        REDIRECT to upload_sample_data.php
        EXIT
    END IF

    // Function to save sample data to database
    DEFINE FUNCTION saveSampleData(connection, data, fileName, fileSize, reportType, metadata)
        LOG "=== SAVE SAMPLE DATA START ==="
        IF data is empty
            LOG error
            RETURN False
        END IF

        TRY
            BEGIN database transaction
            LOG transaction started

            EXTRACT user ID and metadata (dates, account, property)

            INSERT record into CSV_UPLOAD with IsSampleData = 1
            LOG new upload ID

            FOR each row in data
                LOG row data

                GET or CREATE source type in SOURCE_TYPE table

                FOR each metric (sessions, engaged_sessions, users, bounce_rate, avg_session_duration)
                    IF value is valid
                        INSERT data point using insertDataPoint function
                        LOG insertion
                    END IF
                END FOR
            END FOR

            COMMIT transaction
            LOG success

            VERIFY upload and data points saved
            LOG verification

            LOG "=== SAVE SAMPLE DATA SUCCESS ==="
            RETURN True
        CATCH Exception as e
            ROLLBACK transaction
            LOG error and stack trace
            LOG "=== SAVE SAMPLE DATA FAILED ==="
            RETURN False
        END TRY
    END FUNCTION

END





//validate_format.php
START

    REQUIRE admin authentication
    REQUIRE configuration file

    SET response header to JSON

    IF the request method is POST THEN
        GET 'format_name' from POST data, default to empty string if not set
        GET 'detection_columns' from POST data, default to empty string if not set
        GET 'target_fields' from POST data, default to empty string if not set
        GET 'is_new' from POST data, default to 'false' if not set

        INITIALIZE response as a dictionary with 'valid' set to true and 'errors' as an empty list

        // Validate format name
        IF format_name is empty or only whitespace THEN
            SET response['valid'] to false
            ADD "Format name is required" to response['errors']
        ELSE
            IF format_name contains spaces THEN
                SET response['valid'] to false
                ADD "Format name cannot contain spaces. Use underscores instead." to response['errors']
            END IF

            IF format_name contains characters other than letters, numbers, or underscores THEN
                SET response['valid'] to false
                ADD "Format name can only contain letters, numbers, and underscores" to response['errors']
            END IF

            IF format_name is less than 3 characters long THEN
                SET response['valid'] to false
                ADD "Format name must be at least 3 characters long" to response['errors']
            END IF
        END IF

        // Validate detection columns
        IF detection_columns is empty or only whitespace THEN
            SET response['valid'] to false
            ADD "Detection columns are required" to response['errors']
        ELSE
            SPLIT detection_columns by commas, trim each value, and filter out empty values
            IF number of columns is less than 2 THEN
                SET response['valid'] to false
                ADD "At least 2 detection columns are required" to response['errors']
            END IF

            FOR each column in columns
                IF column is empty THEN
                    SET response['valid'] to false
                    ADD "Detection columns cannot be empty" to response['errors']
                    BREAK the loop
                END IF
            END FOR
        END IF

        // Validate target fields for duplicates
        IF target_fields is not empty THEN
            SPLIT target_fields by commas, trim each value, and filter out empty values
            FIND duplicates in the list of target fields
            IF duplicates exist THEN
                SET response['valid'] to false
                ADD "Duplicate target fields detected: ..." to response['errors']
            END IF
        END IF

        // Check if format name already exists (for new formats only)
        IF is_new is 'true' AND format_name is not empty THEN
            SET mappingsFile to the path of csv_mappings.json
            IF mappingsFile exists THEN
                LOAD existingMappings from the JSON file
                CONVERT format_name to lowercase and replace spaces with underscores to get formatKey
                IF formatKey exists in existingMappings THEN
                    SET response['valid'] to false
                    ADD "A format with this name already exists" to response['errors']
                END IF
            END IF
        END IF

        OUTPUT response as JSON

    ELSE
        OUTPUT JSON with 'valid' set to false and error "Invalid request method"
    END IF

END





//migrate_mappings.php
START

    REQUIRE admin authentication
    REQUIRE database configuration

    SET mappingsFile to the path "../config/csv_mappings.json"

    IF mappingsFile does not exist THEN
        OUTPUT "Mappings file not found"
        TERMINATE the script
    END IF

    LOAD mappings from mappingsFile as JSON

    TRY
        BEGIN database transaction

        // Clear existing mappings and formats
        EXECUTE SQL "DELETE FROM column_mapping"
        EXECUTE SQL "DELETE FROM csv_format"
        EXECUTE SQL "ALTER TABLE csv_format AUTO_INCREMENT = 1"
        EXECUTE SQL "ALTER TABLE column_mapping AUTO_INCREMENT = 1"

        SET formatId to 1

        FOR EACH formatKey, format IN mappings
            // Insert CSV format
            SET formatName to formatKey with underscores replaced by spaces and first letter capitalized
            SET reportType to the joined string of format['format_detection'] (comma separated)
            PREPARE SQL statement to INSERT INTO csv_format (FormatID, FormatName, ReportType, AdminUserID, CreatedAt, LastModifiedDate)
            BIND formatId, formatName, reportType to the statement
            EXECUTE the statement

            // Insert column mappings for this format
            FOR EACH csvCol, systemField IN format['column_mappings']
                PREPARE SQL statement to INSERT INTO column_mapping (FormatID, CSVColumnName, SystemFieldName)
                BIND formatId, csvCol, systemField to the statement
                EXECUTE the statement
            END FOR

            INCREMENT formatId by 1
        END FOR

        COMMIT the database transaction

        OUTPUT "Migration completed successfully!"
        OUTPUT "Migrated {number of formats} formats."

    CATCH any Exception AS e
        ROLLBACK the database transaction
        OUTPUT "Migration failed: {error message from e}"
    END TRY

END
