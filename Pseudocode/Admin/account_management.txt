//admin_login.php
START
    START session
    IMPORT config.php

    // Check for admin key in URL
    SET admin_key TO value of GET parameter 'key' OR empty string
    IF admin_key IS NOT 'trafanalyz' THEN
        CALL displayAccessDeniedMessage
    END IF

    // Prevent End-User from accessing admin login
    IF session 'role' IS SET AND session 'role' IS 'End-User' THEN
        CALL displayAccessDeniedMessage
    END IF

    DEFINE FUNCTION displayAccessDeniedMessage
        OUTPUT HTML page with "Access Denied" message and link to homepage
        EXIT script
    END FUNCTION

    INITIALIZE errors AS empty dictionary

    // Redirect if already logged in as admin
    IF session 'user_id' IS SET AND session 'role' IS 'Admin' THEN
        REDIRECT to 'admin/index.php'
        EXIT script
    END IF

    // Handle form submission
    IF request method IS POST THEN
        SET username TO trimmed POST 'username' OR empty string
        SET password TO POST 'password' OR empty string

        // Validate input
        IF username IS empty THEN
            SET errors['username'] TO 'Username is required'
        END IF
        IF password IS empty THEN
            SET errors['password'] TO 'Password is required'
        END IF

        // If no errors, process login
        IF errors IS empty THEN
            PREPARE SQL statement to SELECT UserID, Username, PasswordHash, Role, AccountStatus FROM user WHERE Username = ?
            BIND username TO statement
            EXECUTE statement
            GET result

            IF user record IS found THEN
                IF AccountStatus IS 'Suspended' THEN
                    SET errors['general'] TO 'Your account has been suspended. Please contact support.'
                ELSE IF password matches PasswordHash THEN
                    SET session 'user_id' TO UserID
                    SET session 'username' TO Username
                    SET session 'role' TO Role

                    IF Role IS 'Admin' THEN
                        REDIRECT to 'admin/index.php'
                    ELSE
                        REDIRECT to 'user/index.php'
                    END IF
                    EXIT script
                ELSE
                    SET errors['general'] TO 'Invalid username or password'
                END IF
            ELSE
                SET errors['general'] TO 'Invalid username or password'
            END IF
        END IF
    END IF

    // Render HTML login form
    OUTPUT HTML page:
        - Show error messages if present
        - Login form with username and password fields
        - Password visibility toggle
        - Remember me checkbox
        - Links to admin registration and user login
        - JavaScript for error bubbles and password toggle

END




//admin_register.php
START
    START session
    IMPORT config.php

    SET admin_key TO value from GET parameter 'key' OR empty string
    IF admin_key IS NOT 'trafanalyz' THEN
        CALL displayAccessDeniedMessage
    END IF

    IF session 'role' IS SET AND EQUALS 'End-User' THEN
        CALL displayAccessDeniedMessage
    END IF

    DEFINE FUNCTION displayAccessDeniedMessage
        OUTPUT HTML page with "Access Denied" message and link to homepage
        EXIT script
    END FUNCTION

    INITIALIZE errors AS empty dictionary
    INITIALIZE success AS False

    IF request method IS POST THEN
        SET firstname TO trimmed POST 'firstname' OR empty string
        SET lastname TO trimmed POST 'lastname' OR empty string
        SET username TO trimmed POST 'username' OR empty string
        SET email TO trimmed POST 'email' OR empty string
        SET password TO POST 'password' OR empty string
        SET confirmPassword TO POST 'confirm-password' OR empty string
        SET terms TO True IF POST 'terms' IS SET, ELSE False

        // Validate input
        IF firstname IS empty THEN
            SET errors['firstname'] TO 'First name is required'
        END IF

        IF lastname IS empty THEN
            SET errors['lastname'] TO 'Last name is required'
        END IF

        IF username IS empty THEN
            SET errors['username'] TO 'Username is required'
        ELSE IF length of username < 3 THEN
            SET errors['username'] TO 'Username must be at least 3 characters'
        END IF

        IF email IS empty THEN
            SET errors['email'] TO 'Email is required'
        ELSE IF email IS NOT valid format THEN
            SET errors['email'] TO 'Please enter a valid email address'
        END IF

        IF password IS empty THEN
            SET errors['password'] TO 'Password is required'
        ELSE IF length of password < 8 THEN
            SET errors['password'] TO 'Password must be at least 8 characters'
        END IF

        IF password DOES NOT EQUAL confirmPassword THEN
            SET errors['confirm-password'] TO 'Passwords do not match'
        END IF

        IF terms IS False THEN
            SET errors['terms'] TO 'You must agree to the terms'
        END IF

        // Check for existing users if no errors
        IF errors IS empty THEN
            PREPARE SQL statement to SELECT UserID FROM user WHERE Username = ? OR Email = ?
            BIND username and email to statement
            EXECUTE statement
            IF result has a row THEN
                SET errors['general'] TO 'Username or email already exists'
            ELSE
                HASH password
                PREPARE SQL statement to INSERT INTO user (Username, Email, PasswordHash, Role, AccountStatus) VALUES (?, ?, ?, 'Admin', 'Active')
                BIND username, email, hashed password
                IF statement executes successfully THEN
                    SET success TO True
                    SET session 'register_success' TO True
                ELSE
                    SET errors['general'] TO 'Registration failed. Please try again.'
                END IF
            END IF
        END IF
    END IF

    OUTPUT HTML page:
        - Registration form with fields: firstname, lastname, username, email, password, confirm-password, terms checkbox
        - Show error messages if present
        - Show success popup and redirect if registration is successful
        - JavaScript for client-side validation and password visibility toggle

END

START

    IF session is not started
        START session
    END IF

    DEFINE FUNCTION displayAccessDeniedMessage
        OUTPUT an HTML page with:
            - Title: "Access Denied"
            - Message: "Access denied. Admin area requires proper authorization."
            - Link to return to homepage
        TERMINATE script
    END FUNCTION

    // Authentication checks

    IF 'user_id' is not set in session
        CALL displayAccessDeniedMessage
    END IF

    IF 'role' is not set in session OR 'role' is not equal to 'Admin'
        CALL displayAccessDeniedMessage
    END IF

    IF 'user_id' in session is empty
        CALL displayAccessDeniedMessage
    END IF

    CLOSE session for writing

END




//functions.php
DEFINE FUNCTION getAllUsers WITH PARAMETER conn
    INITIALIZE users as empty list
    TRY
        EXECUTE SQL query to select UserID, Username, Email, Role, AccountStatus, CreatedAt from user table, ordered by UserID
        FOR each row in query result
            ADD row to users list
        END FOR
    CATCH any exception
        LOG error message
    END TRY
    RETURN users
END FUNCTION

DEFINE FUNCTION updateUserStatus WITH PARAMETERS conn, userId, status
    TRY
        IF status is not 'Active' or 'Suspended'
            RETURN False
        END IF
        PREPARE SQL statement to update AccountStatus in user table WHERE UserID = userId
        BIND status and userId to statement
        EXECUTE statement
        RETURN True if successful
    CATCH any exception
        LOG error message
        RETURN False
    END TRY
END FUNCTION

DEFINE FUNCTION deleteUser WITH PARAMETERS conn, userId
    TRY
        BEGIN database transaction
        PREPARE SQL statement to delete from user table WHERE UserID = userId
        BIND userId to statement
        EXECUTE statement
        IF execution successful
            COMMIT transaction
            RETURN True
        ELSE
            ROLLBACK transaction
            RETURN False
        END IF
    CATCH any exception
        ROLLBACK transaction
        LOG error message
        RETURN False
    END TRY
END FUNCTION




//admin_user.php
START

    IMPORT authentication, config, and functions

    INITIALIZE message as empty string
    INITIALIZE messageType as empty string

    IF the request method is POST THEN
        IF 'action' and 'user_id' are set in POST data THEN
            SET userId to integer value of 'user_id'
            SWITCH on 'action'
                CASE 'suspend':
                    IF updateUserStatus succeeds THEN
                        SET message to "User account has been suspended."
                        SET messageType to "success"
                    ELSE
                        SET message to "Failed to suspend user account."
                        SET messageType to "error"
                    ENDIF
                CASE 'restore':
                    IF updateUserStatus succeeds THEN
                        SET message to "User account has been restored."
                        SET messageType to "success"
                    ELSE
                        SET message to "Failed to restore user account."
                        SET messageType to "error"
                    ENDIF
                CASE 'delete':
                    IF deleteUser succeeds THEN
                        SET message to "User account has been deleted."
                        SET messageType to "success"
                    ELSE
                        SET message to "Failed to delete user account."
                        SET messageType to "error"
                    ENDIF
            END SWITCH
        ENDIF
    ENDIF

    // Debug: Count users in database
    EXECUTE SQL to count users in 'user' table
    STORE result in debugCount

    // Retrieve all users
    INITIALIZE users as empty list
    EXECUTE SQL to select user details from 'user' table
    FOR each row in result
        ADD row to users list
    END FOR

    // Render HTML
    DISPLAY admin header

    DISPLAY "User Management" title

    IF message is not empty THEN
        DISPLAY message with messageType
    ENDIF

    DISPLAY debug info: database user count and PHP array user count

    IF users list is empty THEN
        DISPLAY "No users found in the database."
    ELSE
        DISPLAY table of users with columns:
            ID, Username, Email, Role, Status, Created, Actions

        FOR each user in users
            DISPLAY user details in table row
            DISPLAY actions:
                IF user is Active THEN
                    DISPLAY Suspend button
                ELSE
                    DISPLAY Restore button
                ENDIF
                DISPLAY Delete button
        END FOR
    ENDIF

    DISPLAY admin footer

END




//user_management.php
START

    IMPORT authentication_module
    IMPORT config_module
    IMPORT functions_module

    SET pageTitle TO "User Management"

    DISPLAY HTML HEADER WITH pageTitle AND STYLES

    DISPLAY NAVIGATION MENU WITH LINKS:
        - Home
        - Overview
        - User Management (active)
        - CSV Mappings

    DISPLAY MAIN SECTION:
        DISPLAY "User Management" TITLE

        DISPLAY NOTIFICATION AREA (hidden by default)

        DISPLAY SEARCH CONTROLS:
            - Text input for searching users by name, email, or username
            - Search button

        DISPLAY FILTER CONTROLS:
            - All Users
            - Active Users
            - Suspended Users
            - Admins
            - End Users

        DISPLAY USER MANAGEMENT CONTAINER:
            - Show "Loading users..." by default

    DISPLAY FOOTER WITH COPYRIGHT

    LOAD JavaScript MODULE "user-management.js"

    ON DOCUMENT READY:
        CALL UserManagement.initUserManagement WITH 'userManagementContainer'

        DEFINE showNotification(message, type):
            DISPLAY notification with message and style based on type (success/error)
            AUTO-HIDE after 5 seconds

        OVERRIDE UserManagement.suspendUser:
            CALL original suspendUser
            IF success THEN
                CALL showNotification("User has been suspended.", "success")
            ELSE
                CALL showNotification("Failed to suspend user: " + error_message, "error")

        OVERRIDE UserManagement.restoreUser:
            CALL original restoreUser
            IF success THEN
                CALL showNotification("User has been restored.", "success")
            ELSE
                CALL showNotification("Failed to restore user: " + error_message, "error")

        OVERRIDE UserManagement.deleteUser:
            CALL original deleteUser
            IF success THEN
                CALL showNotification("User has been deleted.", "success")
            ELSE
                CALL showNotification("Failed to delete user: " + error_message, "error")

        FOR EACH filter button IN filter-controls:
            ON CLICK:
                SET clicked button as active
                FILTER user rows in userManagementContainer based on:
                    - All: show all
                    - Active: show only users with status "Active"
                    - Suspended: show only users with status "Suspended"
                    - Admin: show only users with role "Admin"
                    - End-User: show only users with role "End-User"

        ON searchButton CLICK OR userSearch ENTER KEY:
            GET search term
            FOR EACH user row:
                IF username, full name, or email CONTAINS search term:
                    SHOW row
                ELSE
                    HIDE row

END




//user_management.js
START

    DEFINE OBJECT UserManagement

        DEFINE FUNCTION suspendUser WITH PARAMETERS userId, callback
            CALL performUserAction WITH 'suspend', userId, callback
        END FUNCTION

        DEFINE FUNCTION restoreUser WITH PARAMETERS userId, callback
            CALL performUserAction WITH 'restore', userId, callback
        END FUNCTION

        DEFINE FUNCTION deleteUser WITH PARAMETERS userId, callback
            IF user confirms deletion
                CALL performUserAction WITH 'delete', userId, callback
            END IF
        END FUNCTION

        DEFINE FUNCTION getAllUsers WITH PARAMETER callback
            CALL performUserAction WITH 'get_users', null, callback
        END FUNCTION

        DEFINE FUNCTION performUserAction WITH PARAMETERS action, userId, callback
            CREATE data dictionary with 'action'
            IF userId is not null
                ADD 'user_id' to data
            END IF

            SEND POST request to 'api_users.php' with data
            IF request is successful
                IF callback is provided
                    CALL callback WITH response data
                END IF
            ELSE
                OUTPUT error to console
                IF callback is provided
                    CALL callback WITH error message
                END IF
            END IF
        END FUNCTION

        DEFINE FUNCTION initUserManagement WITH PARAMETER containerId
            GET container element by containerId
            IF container does not exist
                RETURN
            END IF

            CALL getAllUsers WITH callback
                IF response is successful AND contains data
                    CALL renderUserTable WITH container, response data
                ELSE
                    DISPLAY error message in container
                END IF
            END CALLBACK
        END FUNCTION

        DEFINE FUNCTION renderUserTable WITH PARAMETERS container, users
            IF users list is empty
                DISPLAY "No users found" in container
                RETURN
            END IF

            CREATE HTML table with columns: ID, Username, Full Name, Email, Role, Status, Actions

            FOR EACH user IN users
                ADD row to table with user details
                IF user is Active
                    ADD "Suspend" button
                ELSE
                    ADD "Restore" button
                END IF
                ADD "Delete" button
            END FOR

            SET container's HTML to the table

            FOR EACH "Suspend" button
                ON click, CALL suspendUser WITH userId, callback to reload table or show error
            END FOR

            FOR EACH "Restore" button
                ON click, CALL restoreUser WITH userId, callback to reload table or show error
            END FOR

            FOR EACH "Delete" button
                ON click, CALL deleteUser WITH userId, callback to reload table or show error
            END FOR
        END FUNCTION

    END OBJECT

END





//api_users.php
START

    IMPORT admin authentication
    IMPORT config
    IMPORT functions

    SET response to a dictionary with keys: success (false), message ("No action specified"), data (null)

    IF 'action' and 'user_id' are present in the request
        SET userId to integer value of 'user_id' from request

        SWITCH on 'action' from request
            CASE 'suspend':
                IF updateUserStatus with userId and 'Suspended' returns true
                    SET response to success, message "User account has been suspended", data with userId and status 'Suspended'
                ELSE
                    SET response to failure, message "Failed to suspend user account"
                ENDIF
                BREAK

            CASE 'restore':
                IF updateUserStatus with userId and 'Active' returns true
                    SET response to success, message "User account has been restored", data with userId and status 'Active'
                ELSE
                    SET response to failure, message "Failed to restore user account"
                ENDIF
                BREAK

            CASE 'delete':
                IF deleteUser with userId returns true
                    SET response to success, message "User account has been deleted", data with userId and deleted true
                ELSE
                    SET response to failure, message "Failed to delete user account"
                ENDIF
                BREAK

            CASE 'get_users':
                SET users to result of getAllUsers
                SET response to success, message "Users retrieved successfully", data users
                BREAK

            DEFAULT:
                SET response to failure, message "Invalid action specified"
        ENDSWITCH

    ELSE
        SET response to failure, message "Missing required parameters"
    ENDIF

    OUTPUT response as JSON

END
