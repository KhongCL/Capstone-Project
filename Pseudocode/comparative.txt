    GET userID from session

    INITIALIZE comparison_results, error_message, success_message as null

    // Load user's validated CSV uploads
    PREPARE SQL to select validated uploads for user
    EXECUTE and FETCH uploads

    // Handle file upload and comparison
    IF request method is POST AND both csv_file1 and csv_file2 are uploaded
        GET file1 and file2 from uploaded files

        TRY
            CALL handleCsvUpload for file1 and file2
            IF both uploads are successful
                GET file paths from upload results
                IF both files exist
                    CALL compareCSVFiles with file paths
                    SET success_message
                ELSE
                    SET error_message (files not found)
            ELSE IF both uploads have warnings
                GET file paths from upload results
                IF both files exist
                    CALL compareCSVFiles with file paths
                    SET success_message (with warnings)
                ELSE
                    SET error_message (files not found)
            ELSE
                IF file1 failed
                    SET error_message for file1
                ELSE IF file2 failed
                    SET error_message for file2
                ELSE
                    SET error_message for both files
        CATCH exception
            SET error_message with exception message

    // Handle saving comparison
    IF request method is POST AND 'compare' is set
        GET upload1, upload2, comparisonName from POST
        IF comparisonName is not empty
            INSERT new comparison into saved_comparison table
            INSERT upload1 and upload2 into comparison_file_link table with file order

    // Handle loading saved comparison
    IF request method is POST AND 'load_comparison' is set
        GET selectedComparisonID from POST
        SELECT files for this comparison from DB
        IF at least 2 files found
            SET upload1 and upload2 from results
            CONSTRUCT file paths for both files
            IF both files exist
                TRY
                    CALL compareCSVFiles with file paths
                    SET success_message (loaded comparison)
                CATCH exception
                    SET error_message with exception message
            ELSE
                SET error_message (files missing)
        ELSE
            SET error_message (invalid comparison data)

    // Get user's uploaded CSV files for dropdown
    SELECT all uploads for user, store in csvFiles

    // Get user's saved comparisons for dropdown
    SELECT all saved comparisons for user, store in savedComparisons

    // --- Functions ---

    DEFINE FUNCTION compareCSVFiles(file1_path, file2_path)
        PARSE both CSV files into data1 and data2
        IF either file is empty or invalid
            THROW exception
        GET headers from both files
        DEFINE analytics_metrics list
        INITIALIZE comparison dictionary with basic metrics, headers, analytics_metrics, summary_comparison, data_sample
        FOR each metric in analytics_metrics
            FIND matching column in headers
            IF found
                EXTRACT values from both files
                CLEAN and convert to numeric
                IF both have data
                    CALCULATE stats for both
                    CALCULATE percent change and improvement
                    STORE in comparison['analytics_metrics']
        CALCULATE summary_comparison
        RETURN comparison

    DEFINE FUNCTION findMetricColumn(headers, metric)
        DEFINE metric_variations dictionary
        FOR each variation for the metric
            FOR each header
                IF header matches variation (case-insensitive)
                    RETURN header
        RETURN null

    DEFINE FUNCTION cleanNumericValues(values)
        FOR each value
            REMOVE formatting characters
            IF numeric
                CONVERT to float and add to cleaned list
        RETURN cleaned list

    DEFINE FUNCTION determineImprovement(metric, value2, value1)
        DEFINE higher_is_better and lower_is_better lists
        IF metric in higher_is_better
            RETURN 'improved' if value2 > value1, 'declined' if value2 < value1, else 'unchanged'
        ELSE IF metric in lower_is_better
            RETURN 'improved' if value2 < value1, 'declined' if value2 > value1, else 'unchanged'
        RETURN 'neutral'

    DEFINE FUNCTION calculateSummaryComparison(data1, data2, analytics_metrics)
        FOR each metric in analytics_metrics
            STORE totals, difference, percent change, status in summary
        RETURN summary

    DEFINE FUNCTION parseCSV(file_path)
        OPEN file
        FOR each row
            IF row is metadata or empty, skip
            IF headers not set, set headers
            ELSE IF row matches headers count, add as associative array
        CLOSE file
        RETURN data

    DEFINE FUNCTION calculateStats(values)
        IF values empty, RETURN null
        SORT values
        CALCULATE count, sum, mean, median, min, max, range
        RETURN stats dictionary
